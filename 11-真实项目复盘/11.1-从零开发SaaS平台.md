# 11.1 从零开发 SaaS 平台（完整案例）

> **案例背景**：用 Claude Code 从零开发一个完整的 SaaS 平台（项目管理工具），包含前端、后端、数据库、部署全流程。展示 Claude Code 在真实项目中的完整应用。

---

## 项目概述

**项目名称**：TeamFlow - 团队项目管理 SaaS 平台  
**开发时间**：传统做法 2-3 个月，用 Claude Code **5-7 天**完成核心功能  
**技术栈**：
- 前端：Next.js 14 + TypeScript + Tailwind CSS + Zustand
- 后端：Node.js + Express + Prisma + PostgreSQL
- 部署：Docker + Vercel + Supabase

**核心功能**：
1. 用户认证（注册、登录、SSO）
2. 项目管理（创建、编辑、删除、成员管理）
3. 任务管理（看板、列表、日历视图）
4. 实时协作（WebSocket）
5. 文件上传与分享
6. 数据报表与导出

```javascript
// backend/src/controllers/taskController.js
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

const createTask = async (req, res, next) => {
  try {
    const { title, description, projectId, status, priority, assigneeId, dueDate, columnId } = req.body;
    const userId = req.user.id;

    // 检查项目是否存在且用户是成员
    const projectMember = await prisma.projectMember.findUnique({
      where: {
        userId_projectId: {
          userId,
          projectId,
        },
      },
    });

    if (!projectMember) {
      return res.status(403).json({ success: false, message: '无权访问此项目' });
    }

    // 获取 columnId（如果没提供，则使用项目的第一个列）
    let targetColumnId = columnId;
    if (!targetColumnId) {
      const firstColumn = await prisma.boardColumn.findFirst({
        where: { projectId },
        orderBy: { order: 'asc' },
      });
      if (firstColumn) targetColumnId = firstColumn.id;
    }

    // 获取当前列的最大 order
    const maxOrderTask = await prisma.task.findFirst({
      where: { columnId: targetColumnId },
      orderBy: { order: 'desc' },
    });
    const order = maxOrderTask ? maxOrderTask.order + 1 : 0;

    // 创建任务
    const task = await prisma.task.create({
      data: {
        title,
        description,
        status: status || 'TODO',
        priority: priority || 'MEDIUM',
        dueDate: dueDate ? new Date(dueDate) : null,
        projectId,
        creatorId: userId,
        assigneeId,
        columnId: targetColumnId,
        order,
      },
      include: {
        assignee: { select: { id: true, name: true, avatar: true } },
        creator: { select: { id: true, name: true } },
      },
    });

    // TODO: 发送实时通知（WebSocket）
    // notifyTaskCreated(task);

    res.status(201).json({ success: true, data: task });
  } catch (error) {
    next(error);
  }
};

const updateTaskStatus = async (req, res, next) => {
  try {
    const { id } = req.params;
    const { status, columnId, order } = req.body;
    const userId = req.user.id;

    // 检查任务是否存在且用户有权限
    const task = await prisma.task.findUnique({
      where: { id },
      include: { project: { include: { members: true } } },
    });

    if (!task) {
      return res.status(404).json({ success: false, message: '任务不存在' });
    }

    // 检查权限（只有被分配者或项目管理员可以更新状态）
    const member = task.project.members.find((m) => m.userId === userId);
    if (!member) {
      return res.status(403).json({ success: false, message: '无权操作此任务' });
    }

    if (task.assigneeId !== userId && member.role === 'VIEWER') {
      return res.status(403).json({ success: false, message: '权限不足' });
    }

    // 更新任务状态
    const updatedTask = await prisma.task.update({
      where: { id },
      data: {
        status,
        columnId,
        order,
        updatedAt: new Date(),
      },
      include: {
        assignee: { select: { id: true, name: true, avatar: true } },
        creator: { select: { id: true, name: true } },
      },
    });

    // TODO: 发送实时通知（WebSocket）
    // notifyTaskUpdated(updatedTask);

    res.json({ success: true, data: updatedTask });
  } catch (error) {
    next(error);
  }
};

module.exports = { createTask, getTasks, updateTaskStatus };

import { Task, TaskStatus } from '@/types/task';

interface BoardColumn {
  id: string;
  name: string;
  status: TaskStatus;
  tasks: Task[];
}

const BoardViewPage = ({ params }: { params: { id: string } }) => {
  const projectId = params.id;
  const [columns, setColumns] = useState<BoardColumn[]>([
    { id: 'col-1', name: '待办', status: 'TODO', tasks: [] },
    { id: 'col-2', name: '进行中', status: 'IN_PROGRESS', tasks: [] },
    { id: 'col-3', name: '审核中', status: 'IN_REVIEW', tasks: [] },
    { id: 'col-4', name: '已完成', status: 'DONE', tasks: [] },
  ]);
  const [loading, setLoading] = useState(true);
  const [modalVisible, setModalVisible] = useState(false);
  const [editingTask, setEditingTask] = useState<Task | null>(null);
  const [currentStatus, setCurrentStatus] = useState<TaskStatus>('TODO');

  const fetchTasks = async () => {
    setLoading(true);
    try {
      const tasks = await taskService.getTasks({ projectId });
      
      // 按状态分组
      const grouped = columns.map((col) => ({
        ...col,
        tasks: tasks.filter((task: Task) => task.status === col.status),
      }));
      
      setColumns(grouped);
    } catch (error) {
      console.error('获取任务失败', error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchTasks();
  }, [projectId]);

  const handleDragEnd = async (result: DropResult) => {
    const { source, destination, draggableId } = result;

    // 拖到垃圾桶外，不处理
    if (!destination) return;

    // 同列拖拽
    if (source.droppableId === destination.droppableId) {
      const column = columns.find((col) => col.id === source.droppableId);
      if (!column) return;

      const newTasks = [...column.tasks];
      const [movedTask] = newTasks.splice(source.index, 1);
      newTasks.splice(destination.index, 0, movedTask);

      setColumns((prev) =>
        prev.map((col) =>
          col.id === column.id ? { ...col, tasks: newTasks } : col
        )
      );

      // 更新数据库中的 order
      await taskService.updateTaskOrder({
        taskId: draggableId,
        columnId: column.id,
        order: destination.index,
      });
    } else {
      // 跨列拖拽（更新状态）
      const sourceColumn = columns.find((col) => col.id === source.droppableId);
      const destColumn = columns.find((col) => col.id === destination.droppableId);
      if (!sourceColumn || !destColumn) return;

      // 从源列移除
      const sourceTasks = [...sourceColumn.tasks];
      const [movedTask] = sourceTasks.splice(source.index, 1);

      // 添加到目标列
      const destTasks = [...destColumn.tasks];
      destTasks.splice(destination.index, 0, movedTask);

      setColumns((prev) =>
        prev.map((col) => {
          if (col.id === sourceColumn.id) return { ...col, tasks: sourceTasks };
          if (col.id === destColumn.id) return { ...col, tasks: destTasks };
          return col;
        })
      );

      // 更新数据库中的状态和 order
      await taskService.updateTaskStatus({
        taskId: draggableId,
        status: destColumn.status,
        columnId: destColumn.id,
        order: destination.index,
      });
    }
  };

  const handleCreateTask = (status: TaskStatus) => {
    setCurrentStatus(status);
    setEditingTask(null);
    setModalVisible(true);
  };

  const handleEditTask = (task: Task) => {
    setEditingTask(task);
    setModalVisible(true);
  };

  const handleFormSuccess = () => {
    setModalVisible(false);
    fetchTasks();
  };

  if (loading) return <Spin size="large" />;

  return (
    <div className="h-screen flex flex-col">
      <div className="flex-1 flex overflow-x-auto p-4 space-x-4">
        <DragDropContext onDragEnd={handleDragEnd}>
          {columns.map((column) => (
            <div key={column.id} className="flex-shrink-0 w-80">
              <div className="bg-gray-100 rounded-lg p-4">
                <div className="flex justify-between items-center mb-4">
                  <h3 className="font-semibold text-gray-700">
                    {column.name}{' '}
                    <span className="text-gray-400">({column.tasks.length})</span>
                  </h3>
                  <Button
                    type="text"
                    icon={<PlusOutlined />}
                    onClick={() => handleCreateTask(column.status)}
                  />
                </div>

                <Droppable droppableId={column.id}>
                  {(provided, snapshot) => (
                    <div
                      ref={provided.innerRef}
                      {...provided.droppableProps}
                      className={`space-y-2 min-h-[200px] ${
                        snapshot.isDraggingOver ? 'bg-blue-50' : ''
                      }`}
                    >
                      {column.tasks.map((task, index) => (
                        <Draggable
                          key={task.id}
                          draggableId={task.id}
                          index={index}
                        >
                          {(provided, snapshot) => (
                            <div
                              ref={provided.innerRef}
                              {...provided.draggableProps}
                              {...provided.dragHandleProps}
                              className={snapshot.isDragging ? 'opacity-50' : ''}
                            >
                              <TaskCard task={task} onClick={() => handleEditTask(task)} />
                            </div>
                          )}
                        </Draggable>
                      )}
                      {provided.placeholder}
                    </div>
                  )}
                </Droppable>
              </div>
            </div>
          ))}
        </DragDropContext>
      </div>

      <Modal
        title={editingTask ? '编辑任务' : '创建任务'}
        open={modalVisible}
        onCancel={() => setModalVisible(false)}
        footer={null}
        width={600}
      >
        <TaskForm
          projectId={projectId}
          initialValues={editingTask}
          status={currentStatus}
          onSuccess={handleFormSuccess}
        />
      </Modal>
    </div>
  );
};

export default BoardViewPage;
```
