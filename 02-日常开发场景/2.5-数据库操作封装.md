# 2.5 数据库操作封装

## 场景描述

每个项目都要操作数据库：连接、查询、迁移、种子数据……

但手写 SQL 容易出错，而且不同数据库的语法不同。

**Claude Code 可以帮你生成统一的数据库操作层，支持多种数据库。**

---

## 案例：博客系统数据库层

我们要为博客系统封装数据库操作，支持：
- 数据库连接管理（连接池）
- CRUD 操作封装
- 数据库迁移（Migration）
- 种子数据（Seeds）
- 支持 PostgreSQL 和 SQLite（开发环境用 SQLite，生产用 PostgreSQL）

### src/models/Post.js（文章模型）

```javascript
const db = require('../db');

class Post {
  // 创建文章
  static async create({ title, content, author_id, category_id, status = 'draft' }) {
    const [post] = await db('posts')
      .insert({ title, content, author_id, category_id, status })
      .returning('*');
    return post;
  }
  
  // 查找文章（带关联数据）
  static async findById(id) {
    const post = await db('posts')
      .select(
        'posts.*',
        'users.name as author_name',
        'categories.name as category_name'
      )
      .leftJoin('users', 'posts.author_id', 'users.id')
      .leftJoin('categories', 'posts.category_id', 'categories.id')
      .where('posts.id', id)
      .first();
    
    return post;
  }
  
  // 获取文章列表（支持分页、筛选、搜索）
  static async findAll({ page = 1, limit = 10, category, status, search } = {}) {
    const offset = (page - 1) * limit;
    
    let query = db('posts')
      .select(
        'posts.*',
        'users.name as author_name',
        'categories.name as category_name'
      )
      .leftJoin('users', 'posts.author_id', 'users.id')
      .leftJoin('categories', 'posts.category_id', 'categories.id');
    
    if (category) {
      query = query.where('posts.category_id', category);
    }
    
    if (status) {
      query = query.where('posts.status', status);
    }
    
    if (search) {
      query = query.where(function() {
        this.where('posts.title', 'like', `%${search}%`)
          .orWhere('posts.content', 'like', `%${search}%`);
      });
    }
    
    const total = await query.clone().count('* as count').first();
    const posts = await query.orderBy('created_at', 'desc').limit(limit).offset(offset);
    
    return {
      data: posts,
      pagination: {
        page,
        limit,
        total: total.count,
        totalPages: Math.ceil(total.count / limit)
      }
    };
  }
  
  // 更新文章
  static async update(id, { title, content, category_id, status }) {
    const [post] = await db('posts')
      .where({ id })
      .update({ title, content, category_id, status, updated_at: db.fn.now() })
      .returning('*');
    return post;
  }
  
  // 删除文章（软删除）
  static async delete(id) {
    const count = await db('posts')
      .where({ id })
      .update({ deleted_at: db.fn.now() });
    return count > 0;
  }
  
  // 获取文章统计
  static async getStats(author_id) {
    const stats = await db('posts')
      .select('status')
      .count('* as count')
      .where({ author_id })
      .whereNull('deleted_at')
      .groupBy('status');
    
    return stats;
  }
}

module.exports = Post;
```

exports.up = function(knex) {
  return knex.schema.createTable('users', (table) => {
    table.increments('id').primary();
    table.string('username', 50).unique().notNullable();
    table.string('email', 100).unique().notNullable();

    table.timestamp('updated_at').defaultTo(knex.fn.now());

  });
};

### src/migrations/002_create_posts.js（文章表迁移）

```javascript
exports.up = function(knex) {
  return knex.schema.createTable('posts', (table) => {
    table.increments('id').primary();
    table.string('title', 200).notNullable();
    table.text('content').notNullable();
    table.integer('author_id').unsigned().notNullable();
    table.integer('category_id').unsigned();
    table.string('status', 20).notNullable().defaultTo('draft');
    table.timestamp('created_at').defaultTo(knex.fn.now());
    table.timestamp('updated_at').defaultTo(knex.fn.now());
    table.timestamp('deleted_at').nullable(); // 软删除
    
    // 外键
    table.foreign('author_id').references('users.id').onDelete('CASCADE');
    table.foreign('category_id').references('categories.id').onDelete('SET NULL');
    
    // 索引
    table.index(['author_id', 'status', 'category_id']);
    table.index(['created_at']);
  });
};

exports.down = function(knex) {
  return knex.schema.dropTable('posts');
};
```

## 使用 Migration 和 Seeds

### 运行迁移

```bash
# 安装 Knex CLI
npm install -g knex

# 运行所有迁移
knex migrate:latest

# 回滚上一次迁移
knex migrate:rollback

# 回滚所有迁移
knex migrate:rollback --all
```

### 插入种子数据

```bash
# 运行所有种子
knex seed:run

# 运行指定种子
knex seed:run --specific=users.js
```

## 最佳实践

### 1. 用 Migration 管理数据库变更

❌ **错误做法：**
```sql
-- 直接手动改数据库
ALTER TABLE posts ADD COLUMN views INT;
```

✅ **正确做法：**
```javascript
// 创建迁移文件
exports.up = function(knex) {
  return knex.schema.table('posts', (table) => {
    table.integer('views').defaultTo(0);
  });
};
```

---

### 2. 用 Seeds 初始化测试数据

```javascript
// 开发环境和测试环境用种子数据
if (process.env.NODE_ENV !== 'production') {
  await knex.seed.run();
}
```

---

### 3. 模型方法要覆盖常见操作

至少要有：
- `create`（创建）
- `findById`（查找单条）
- `findAll`（查找多条，支持分页和筛选）
- `update`（更新）
- `delete`（删除）
- `count`（统计）

---

### 4. 连接池配置要合理

```javascript
pool: {
  min: 2,      // 最小连接数
  max: 10,      // 最大连接数（生产环境建议 50+）
  acquireTimeoutMillis: 30000, // 获取连接超时
  createTimeoutMillis: 30000,  // 创建连接超时
  idleTimeoutMillis: 30000,    // 空闲连接超时
  reapIntervalMillis: 1000,    // 清理间隔
  createRetryIntervalMillis: 100 // 创建重试间隔
}
```

## 关键要点

1. **用 ORM/查询构建器**：避免手写 SQL，提升可维护性
2. **Migration 管理变更**：数据库版本控制，方便回滚
3. **Seeds 初始化数据**：开发/测试环境快速填充数据
4. **连接池配置合理**：避免数据库连接耗尽
5. **模型方法要完整**：覆盖 CRUD + 统计 + 关联查询
