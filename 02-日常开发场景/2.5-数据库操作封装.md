### 第三步：核心代码解读

#### 1. 数据库连接配置（`config/database.js`）

```javascript
const mongoose = require('mongoose');
const logger = require('./logger');

const connectDB = async () => {
  try {
    const uri = process.env.MONGODB_URI || 'mongodb://localhost:27017/todoapp';
    
    await mongoose.connect(uri, {
      useNewUrlParser: true,
      useUnifiedTopology: true,
    });
    
    logger.info('✅ MongoDB 连接成功', { uri: uri.replace(/\/\/.*@/, '//***:***@') });  // 隐藏密码
    
    // 监听连接事件
    mongoose.connection.on('error', (err) => {
      logger.error('MongoDB 连接错误', { error: err.message });
    });
    
    mongoose.connection.on('disconnected', () => {
      logger.warn('MongoDB 断开连接，尝试重连...');
      setTimeout(connectDB, 5000);  // 5秒后重连
    });
    
  } catch (error) {
    logger.error('MongoDB 连接失败', { error: error.message });
    process.exit(1);  // 连接失败，退出进程
  }
};

module.exports = connectDB;
```

**使用方式**（在 `server.js` 中）：

```javascript
const connectDB = require('./config/database');

// 连接数据库
connectDB();

// 启动服务器
app.listen(PORT, () => {
  console.log(`🚀 服务器运行在 http://localhost:${PORT}`);
});
```

#### 2. Mongoose Schema 定义（`models/Todo.js`）

```javascript
const mongoose = require('mongoose');

const TodoSchema = new mongoose.Schema({
  title: {
    type: String,
    required: [true, '标题不能为空'],
    minlength: [1, '标题至少 1 个字符'],
    maxlength: [200, '标题最多 200 个字符']
  },
  description: {
    type: String,
    default: ''
  },
  completed: {
    type: Boolean,
    default: false
  }
}, {
  timestamps: true  // 自动添加 createdAt 和 updatedAt
});

// 实例方法：标记为完成
TodoSchema.methods.markComplete = function() {
  this.completed = true;
  return this.save();
};

// 静态方法：查找所有未完成的待办
TodoSchema.statics.findIncomplete = function() {
  return this.find({ completed: false });
};

const Todo = mongoose.model('Todo', TodoSchema);

module.exports = Todo;
```

**好处**：
- **数据校验**：在 Schema 层校验数据合法性
- **实例方法**：定义在单个文档上的方法（如 `todo.markComplete()`）
- **静态方法**：定义在模型上的方法（如 `Todo.findIncomplete()`）

#### 3. 数据操作层 DAO（`dao/todoDao.js`）

```javascript
const Todo = require('../models/Todo');
const logger = require('../utils/logger');

const todoDao = {
  // 创建待办事项
  create: async ({ title, description }) => {
    try {
      const todo = new Todo({ title, description });
      const saved = await todo.save();
      logger.info('待办事项创建成功', { id: saved._id });
      return saved;
    } catch (error) {
      logger.error('待办事项创建失败', { error: error.message });
      throw error;
    }
  },

  // 获取所有待办事项
  findAll: async () => {
    try {
      const todos = await Todo.find().sort({ createdAt: -1 });  // 按创建时间倒序
      return todos;
    } catch (error) {
      logger.error('获取待办事项失败', { error: error.message });
      throw error;
    }
  },

  // 根据 ID 获取单个待办事项
  findById: async (id) => {
    try {
      const todo = await Todo.findById(id);
      return todo;
    } catch (error) {
      logger.error('查找待办事项失败', { id, error: error.message });
      throw error;
    }
  },

  // 更新待办事项
  update: async (id, updates) => {
    try {
      const todo = await Todo.findByIdAndUpdate(
        id,
        updates,
        { new: true, runValidators: true }  // 返回更新后的文档，运行校验
      );
      if (todo) {
        logger.info('待办事项更新成功', { id });
      }
      return todo;
    } catch (error) {
      logger.error('待办事项更新失败', { id, error: error.message });
      throw error;
    }
  },

  // 删除待办事项
  delete: async (id) => {
    try {
      const todo = await Todo.findByIdAndDelete(id);
      if (todo) {
        logger.info('待办事项删除成功', { id });
      }
      return todo;
    } catch (error) {
      logger.error('待办事项删除失败', { id, error: error.message });
      throw error;
    }
  }
};

module.exports = todoDao;
```

**好处**：
- **错误处理**：统一的错误日志和异常抛出
- **数据操作集中**：所有数据库操作在一个文件里
- **易于替换**：后续换数据库只需修改 DAO 层

```javascript
const todoDao = require('../dao/todoDao');
const logger = require('../utils/logger');

const create = async ({ title, description }) => {
  const newTodo = await todoDao.create({ title, description });
  return newTodo;

const getAll = async () => {
  const todos = await todoDao.findAll();
  return todos;
};

const getById = async (id) => {
  const todo = await todoDao.findById(id);
  return todo;
};

  const updated = await todoDao.update(id, updates);
  return updated;
};

const remove = async (id) => {
  const deleted = await todoDao.delete(id);
  return deleted;
};

module.exports = { create, getAll, getById, update, remove };
```

### 2. 用 DAO 模式封装数据库操作

❌ **不好**：在 Service 层直接调用 `Todo.find()`
✅ **好**：用 DAO 层封装

**好处**：
- **隔离变化**：后续换数据库只需修改 DAO 层
- **代码复用**：多个 Service 可以共用一个 DAO
- **易于测试**：可以用 mock DAO 进行单元测试

---

### 3. Schema 层数据校验

用 Mongoose Schema 的 `required`、`minlength`、`maxlength` 等校验规则，防止脏数据进入数据库。

**示例**：

```javascript
const TodoSchema = new mongoose.Schema({
  title: {
    type: String,
    required: [true, '标题不能为空'],
    minlength: [1, '标题至少 1 个字符'],
    maxlength: [200, '标题最多 200 个字符']
  }
});
```

---

### 4. 数据库连接错误处理

❌ **不好**：连接失败不处理，服务器直接崩溃
✅ **好**：监听 `error` 和 `disconnected` 事件，自动重连
