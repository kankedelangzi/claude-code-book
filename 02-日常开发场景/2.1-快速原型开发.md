`routes/todos.js`：
```javascript
const express = require('express');
const { getTodos, getTodoById, createTodo, updateTodo, deleteTodo } = require('../data/store');

const router = express.Router();

// 获取所有待办事项
router.get('/', (req, res) => {
  const todos = getTodos();
  res.json({ success: true, data: todos });
});

// 获取单个待办事项
router.get('/:id', (req, res) => {
  const todo = getTodoById(req.params.id);
  if (!todo) {
    return res.status(404).json({ success: false, error: '待办事项不存在' });
  }
  res.json({ success: true, data: todo });
});

// 创建待办事项
router.post('/', (req, res) => {
  const { title, description } = req.body;
  
  if (!title) {
    return res.status(400).json({ success: false, error: '标题不能为空' });
  }
  
  const newTodo = createTodo({ title, description });
  res.status(201).json({ success: true, data: newTodo });
});

// 更新待办事项
router.put('/:id', (req, res) => {
  const { title, description, completed } = req.body;
  const updated = updateTodo(req.params.id, { title, description, completed });
  
  if (!updated) {
    return res.status(404).json({ success: false, error: '待办事项不存在' });
  }
  
  res.json({ success: true, data: updated });
});

// 删除待办事项
router.delete('/:id', (req, res) => {
  const deleted = deleteTodo(req.params.id);
  
  if (!deleted) {
    return res.status(404).json({ success: false, error: '待办事项不存在' });
  }
  
  res.json({ success: true, message: '删除成功' });
});

module.exports = router;
```

`data/store.js`：
```javascript
// 内存存储（原型阶段用这个，简单快速）
let todos = [];
let nextId = 1;

const getTodos = () => {
  return todos;
};

const getTodoById = (id) => {
  return todos.find(todo => todo.id === id);
};

const createTodo = ({ title, description }) => {
  const newTodo = {
    id: String(nextId++),
    title,
    description: description || '',
    completed: false,
    createdAt: new Date().toISOString()
  };
  todos.push(newTodo);
  return newTodo;
};

const updateTodo = (id, updates) => {
  const index = todos.findIndex(todo => todo.id === id);
  if (index === -1) return null;
  
  todos[index] = { ...todos[index], ...updates };
  return todos[index];
};

const deleteTodo = (id) => {
  const index = todos.findIndex(todo => todo.id === id);
  if (index === -1) return null;
  
  const deleted = todos[index];
  todos = todos.filter(todo => todo.id !== id);
  return deleted;
};

module.exports = { getTodos, getTodoById, createTodo, updateTodo, deleteTodo };
```

## 核心技巧

### 1. 需求表达要"刚好够用"

❌ **不好**：我想做一个类似 Trello 的项目管理工具，支持多人协作、实时同步、权限管理...
✅ **好**：我想做一个待办 API，支持增删改查，数据存内存，返回 JSON。

**原因**：原型阶段过度设计是浪费时间。先跑起来，再迭代。

---

### 2. 让 Claude Code 生成项目结构

不要自己手动建文件夹，直接说：

```
请帮我生成项目结构，包含：
- server.js（入口文件）
- routes/（路由）
- data/（数据存储）
- 合理的 .gitignore
```

Claude Code 会一次性生成所有文件，比你手动建快 10 倍。

---

### 3. 原型阶段用内存存储

❌ **不好**：一开始就配置 MongoDB/PostgreSQL。
✅ **好**：用数组/对象存内存，跑通流程后再换数据库。

**原因**：
- 内存存储零配置，立即能用
- 方便测试（重启即清空）
- 数据库配置可以后面让 Claude Code 帮你加

---

### 4. 快速验证用 curl，别搞前端

原型阶段不需要漂亮的前端界面，用 curl 或 Postman 测试 API 就够了。

如果你想让非技术人员也能测试，可以让 Claude Code 生成一个简单的 HTML 测试页面：

```
帮我生成一个简单的 HTML 页面，能调用这个 API 进行测试。
```
