## 实战：10 分钟搭建待办 API

### 第一步：创建项目（30 秒）

```bash
mkdir todo-api && cd todo-api
npm init -y
npm install express cors body-parser
```

### 第二步：让 Claude Code 生成核心代码

打开 Claude Code，输入：

```
帮我创建一个待办事项 API，要求：
1. RESTful 风格（GET/POST/PUT/DELETE）
2. 数据存在 JSON 文件（不用数据库）
3. 支持按状态筛选（all/active/completed）
4. 包含基础错误处理
5. 代码要简洁，方便演示

文件结构：
- server.js（主入口）
- todoController.js（业务逻辑）
- data/todos.json（数据存储）
```

**Claude Code 输出：**

`'todoController.js'`

```javascript
const fs = require('fs').promises;
const path = require('path');

const DATA_FILE = path.join(__dirname, '../data/todos.json');

// 读取所有待办
async function getTodos(filter = 'all') {
  const data = await fs.readFile(DATA_FILE, 'utf8');
  const todos = JSON.parse(data);
  
  if (filter === 'active') return todos.filter(t => !t.completed);
  if (filter === 'completed') return todos.filter(t => t.completed);
  return todos;
}

// 创建待办
async function createTodo(title) {
  const data = await fs.readFile(DATA_FILE, 'utf8');
  const todos = JSON.parse(data);
  
  const newTodo = {
    id: Date.now().toString(),
    title,
    completed: false,
    createdAt: new Date().toISOString()
  };
  
  todos.push(newTodo);
  await fs.writeFile(DATA_FILE, JSON.stringify(todos, null, 2));
  return newTodo;
}

// 更新待办
async function updateTodo(id, updates) {
  const data = await fs.readFile(DATA_FILE, 'utf8');
  const todos = JSON.parse(data);
  
  const index = todos.findIndex(t => t.id === id);
  if (index === -1) throw new Error('Todo not found');
  
  todos[index] = { ...todos[index], ...updates };
  await fs.writeFile(DATA_FILE, JSON.stringify(todos, null, 2));
  return todos[index];
}

// 删除待办
async function deleteTodo(id) {
  const data = await fs.readFile(DATA_FILE, 'utf8');
  const todos = JSON.parse(data);
  
  const filtered = todos.filter(t => t.id !== id);
  if (filtered.length === todos.length) throw new Error('Todo not found');
  
  await fs.writeFile(DATA_FILE, JSON.stringify(filtered, null, 2));
  return { success: true };
}

module.exports = { getTodos, createTodo, updateTodo, deleteTodo };
```

`'server.js'`

```javascript
const express = require('express');
const cors = require('cors');
const bodyParser = require('body-parser');
const { getTodos, createTodo, updateTodo, deleteTodo } = require('./todoController');

const app = express();
const PORT = 3000;

// 中间件
app.use(cors());
app.use(bodyParser.json());

// 初始化数据文件
const fs = require('fs');
const path = require('path');
const DATA_DIR = path.join(__dirname, 'data');
const DATA_FILE = path.join(DATA_DIR, 'todos.json');

if (!fs.existsSync(DATA_DIR)) {
  fs.mkdirSync(DATA_DIR);
}
if (!fs.existsSync(DATA_FILE)) {
  fs.writeFileSync(DATA_FILE, '[]');
}

// 路由
app.get('/api/todos', async (req, res) => {
  try {
    const { filter } = req.query;
    const todos = await getTodos(filter);
    res.json(todos);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.post('/api/todos', async (req, res) => {
  try {
    const { title } = req.body;
    if (!title) return res.status(400).json({ error: 'Title is required' });
    
    const todo = await createTodo(title);
    res.status(201).json(todo);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.put('/api/todos/:id', async (req, res) => {
  try {
    const { id } = req.params;
    const updates = req.body;
    const todo = await updateTodo(id, updates);
    res.json(todo);
  } catch (error) {
    res.status(404).json({ error: error.message });
  }
});

app.delete('/api/todos/:id', async (req, res) => {
  try {
    const { id } = req.params;
    await deleteTodo(id);
    res.json({ success: true });
  } catch (error) {
    res.status(404).json({ error: error.message });
  }
});

app.listen(PORT, () => {
  console.log(`✅ Todo API running on http://localhost:${PORT}`);
});
```

## 关键要点

| 技巧 | 说明 |
|------|------|
| **明确需求** | 告诉 Claude Code 你的具体需求（RESTful、JSON 存储、筛选） |
| **指定文件结构** | 提前规划好文件分工，避免代码全堆在一个文件 |
| **要求错误处理** | 原型也要有基本的 error handling，方便演示 |
| **可运行优先** | 先让代码跑起来，再优化细节 |

---

## 进阶：让原型更专业

如果想让 demo 更拿得出手，可以继续让 Claude Code 添加：

1. **输入验证**（用 Joi 或 express-validator）
2. **API 文档**（生成 OpenAPI/Swagger）
3. **单元测试**（用 Jest 写几个测试用例）
4. **日志**（用 winston 或 morgan）

示例指令：

```
给这个 API 加上：
1. 用 express-validator 验证输入
2. 用 Jest 写单元测试
3. 添加 Swagger API 文档

保持代码简洁，适合演示用。
```

## 小结

| 传统方式 | Claude Code 方式 |
|---------|-----------------|
| 查文档、写 boilerplate | 直接描述需求，自动生成 |
| 手动调试路由、错误处理 | 一次性生成可运行代码 |
| 1-2 小时 | 10 分钟 |

**核心思想**：原型开发的目标是**快速验证想法**，不是写出生产级代码。Claude Code 帮你把重复性工作自动化，你专注于业务逻辑和创意。
