# 2.3 前端组件生成

## 场景描述

前端开发日常：写组件、调样式、对接 API。

但很多组件是重复劳动：表单、列表、弹窗、搜索框……

**Claude Code 可以帮你秒生成可复用的前端组件。**

---

## 案例：用户管理界面（React）

我们要做一个用户管理页面，包含：
- 用户列表（带分页、搜索）
- 创建/编辑用户的弹窗表单

- 对接后端 API

## 核心代码解析

### src/App.jsx（主应用）

```jsx
import React, { useState, useEffect } from 'react';
import UserList from './components/UserList';
import UserForm from './components/UserForm';
import SearchBar from './components/SearchBar';
import { getUsers, createUser, updateUser, deleteUser } from './services/userService';

function App() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const [search, setSearch] = useState('');
  const [page, setPage] = useState(1);
  const [totalPages, setTotalPages] = useState(1);
  const [showForm, setShowForm] = useState(false);
  const [editingUser, setEditingUser] = useState(null);

  // 加载用户列表
  const fetchUsers = async () => {
    setLoading(true);
    setError(null);
    try {
      const data = await getUsers({ search, page, limit: 10 });
      setUsers(data.users);
      setTotalPages(data.totalPages);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchUsers();
  }, [search, page]);

  // 创建用户
  const handleCreate = async (userData) => {
    setLoading(true);
    try {
      await createUser(userData);
      setShowForm(false);
      fetchUsers();
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  // 更新用户
  const handleUpdate = async (id, userData) => {
    setLoading(true);
    try {
      await updateUser(id, userData);
      setShowForm(false);
      setEditingUser(null);
      fetchUsers();
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  // 删除用户
  const handleDelete = async (id) => {
    if (!window.confirm('确定要删除这个用户吗？')) return;
    
    setLoading(true);
    try {
      await deleteUser(id);
      fetchUsers();
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen bg-gray-100 p-8">
      <div className="max-w-6xl mx-auto">
        <h1 className="text-3xl font-bold mb-8">用户管理</h1>

        {/* 搜索和创建按钮 */}
        <div className="flex justify-between mb-6">
          <SearchBar onSearch={setSearch} />
          <button
            onClick={() => {
              setEditingUser(null);
              setShowForm(true);
            }}
            className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
          >
            + 新建用户
          </button>
        </div>

        {/* 错误提示 */}
        {error && (
          <div className="bg-red-100 border border-red-400 text-red-700 px-4 py-3 rounded mb-4">
            {error}
          </div>
        )}

        {/* 加载状态 */}
        {loading && (
          <div className="text-center py-8">
            <div className="spinner"></div>
          </div>
        )}

        {/* 用户列表 */}
        {!loading && (
          <UserList
            users={users}
            onEdit={(user) => {
              setEditingUser(user);
              setShowForm(true);
            }}
            onDelete={handleDelete}
          />
        )}

        {/* 分页 */}
        {!loading && users.length > 0 && (
          <div className="flex justify-center mt-6">
            <button
              onClick={() => setPage(p => Math.max(1, p - 1))}
              disabled={page === 1}
              className="px-4 py-2 bg-white border rounded-l disabled:opacity-50"
            >
              上一页
            </button>
            <span className="px-4 py-2 bg-white border-t border-b">
              {page} / {totalPages}
            </span>
            <button
              onClick={() => setPage(p => Math.min(totalPages, p + 1))}
              disabled={page === totalPages}
              className="px-4 py-2 bg-white border rounded-r disabled:opacity-50"
            >
              下一页
            </button>
          </div>
        )}

        {/* 空状态 */}
        {!loading && users.length === 0 && (
          <div className="text-center py-8 text-gray-500">
            暂无用户数据
          </div>
        )}

        {/* 创建/编辑表单弹窗 */}
        {showForm && (
          <UserForm
            user={editingUser}
            onSubmit={editingUser ? (data) => handleUpdate(editingUser.id, data) : handleCreate}
            onCancel={() => {
              setShowForm(false);
              setEditingUser(null);
            }}
          />
        )}
      </div>
    </div>
  );
}

export default App;
```

### src/components/UserForm.jsx（创建/编辑表单）

```jsx
import React, { useState } from 'react';

function UserForm({ user, onSubmit, onCancel }) {
  const [formData, setFormData] = useState({
    name: user?.name || '',
    email: user?.email || '',
    role: user?.role || 'user',
    password: ''
  });
  const [errors, setErrors] = useState({});

  // 表单验证
  const validate = () => {
    const newErrors = {};
    if (!formData.name) newErrors.name = '姓名不能为空';
    if (!formData.email) newErrors.email = '邮箱不能为空';
    else if (!/\S+@\S+\.\S+/.test(formData.email)) newErrors.email = '邮箱格式不正确';
    if (!user && !formData.password) newErrors.password = '密码不能为空';
    return newErrors;
  };

  // 提交表单
  const handleSubmit = (e) => {
    e.preventDefault();
    const newErrors = validate();
    if (Object.keys(newErrors).length > 0) {
      setErrors(newErrors);
      return;
    }
    onSubmit(formData);
  };

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex justify-center items-center">
      <div className="bg-white rounded-lg p-6 w-full max-w-md">
        <h2 className="text-2xl font-bold mb-4">
          {user ? '编辑用户' : '新建用户'}
        </h2>

        <form onSubmit={handleSubmit}>
          {/* 姓名 */}
          <div className="mb-4">
            <label className="block text-sm font-medium text-gray-700 mb-2">
              姓名
            </label>
            <input
              type="text"
              value={formData.name}
              onChange={(e) => setFormData({ ...formData, name: e.target.value })}
              className={`w-full px-3 py-2 border rounded-lg ${
                errors.name ? 'border-red-500' : 'border-gray-300'
              }`}
            />
            {errors.name && (
              <p className="text-red-500 text-xs mt-1">{errors.name}</p>
            )}
          </div>

          {/* 邮箱 */}
          <div className="mb-4">
            <label className="block text-sm font-medium text-gray-700 mb-2">
              邮箱
            </label>
            <input
              type="email"
              value={formData.email}
              onChange={(e) => setFormData({ ...formData, email: e.target.value })}
              className={`w-full px-3 py-2 border rounded-lg ${
                errors.email ? 'border-red-500' : 'border-gray-300'
              }`}
            />
            {errors.email && (
              <p className="text-red-500 text-xs mt-1">{errors.email}</p>
            )}
          </div>

          {/* 角色 */}
          <div className="mb-4">
            <label className="block text-sm font-medium text-gray-700 mb-2">
              角色
            </label>
            <select
              value={formData.role}
              onChange={(e) => setFormData({ ...formData, role: e.target.value })}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg"
            >
              <option value="user">普通用户</option>
              <option value="admin">管理员</option>
            </select>
          </div>

          {/* 密码 */}
          {!user && (
            <div className="mb-4">
              <label className="block text-sm font-medium text-gray-700 mb-2">
                密码
              </label>
              <input
                type="password"
                value={formData.password}
                onChange={(e) => setFormData({ ...formData, password: e.target.value })}
                className={`w-full px-3 py-2 border rounded-lg ${
                  errors.password ? 'border-red-500' : 'border-gray-300'
                }`}
              />
              {errors.password && (
                <p className="text-red-500 text-xs mt-1">{errors.password}</p>
              )}
            </div>
          )}

          {/* 按钮 */}
          <div className="flex justify-end gap-4">
            <button
              type="button"
              onClick={onCancel}
              className="px-4 py-2 border border-gray-300 rounded-lg hover:bg-gray-50"
            >
              取消
            </button>
            <button
              type="submit"
              className="px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600"
            >
              {user ? '更新' : '创建'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

export default UserForm;
```

### src/services/userService.js（API 调用封装）

```javascript
import axios from 'axios';

const API_BASE = '/api/users';

// 获取用户列表
export const getUsers = async ({ search, page, limit }) => {
  const params = new URLSearchParams();
  if (search) params.append('search', search);
  if (page) params.append('page', page);
  if (limit) params.append('limit', limit);
  
  const response = await axios.get(`${API_BASE}?${params.toString()}`);
  return response.data;
};

// 获取单个用户
export const getUser = async (id) => {
  const response = await axios.get(`${API_BASE}/${id}`);
  return response.data;
};

// 创建用户
export const createUser = async (userData) => {
  const response = await axios.post(API_BASE, userData);
  return response.data;
};

// 更新用户
export const updateUser = async (id, userData) => {
  const response = await axios.put(`${API_BASE}/${id}`, userData);
  return response.data;
};

// 删除用户
export const deleteUser = async (id) => {
  const response = await axios.delete(`${API_BASE}/${id}`);
  return response.data;
};
```

## 关键要点

1. **组件拆分要合理**：列表、表单、搜索、弹窗各自独立
2. **状态管理要清晰**：用 `useState` + `useEffect` 管理数据和加载状态
3. **错误处理不可少**：API 调用必须 try-catch
4. **加载状态和空状态**：提升用户体验
5. **表单验证**：前端验证 + 后端验证双保险
