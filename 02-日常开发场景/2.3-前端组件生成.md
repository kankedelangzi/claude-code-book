### 第三步：核心代码解读

#### 1. API 调用层（`services/todoService.js`）

```javascript
const API_BASE = 'http://localhost:3000/api/todos';

const todoService = {
  // 获取所有待办事项
  getAll: async () => {
    const response = await fetch(API_BASE);
    if (!response.ok) throw new Error('获取待办事项失败');
    const result = await response.json();
    return result.data;
  },

  // 创建待办事项
  create: async (title, description) => {
    const response = await fetch(API_BASE, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ title, description })
    });
    if (!response.ok) throw new Error('创建待办事项失败');
    const result = await response.json();
    return result.data;
  },

  // 更新待办事项
  update: async (id, updates) => {
    const response = await fetch(`${API_BASE}/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(updates)
    });
    if (!response.ok) throw new Error('更新待办事项失败');
    const result = await response.json();
    return result.data;
  },

  // 删除待办事项
  delete: async (id) => {
    const response = await fetch(`${API_BASE}/${id}`, {
      method: 'DELETE'
    });
    if (!response.ok) throw new Error('删除待办事项失败');
    return true;
  }
};

export default todoService;
```

**好处**：
- API 调用集中管理，方便维护
- 错误处理统一，组件里不用重复写 `try-catch`

#### 2. 待办列表组件（`components/TodoList.js`）

```javascript
import { useState, useEffect } from 'react';
import todoService from '../services/todoService';
import TodoItem from './TodoItem';
import TodoForm from './TodoForm';

const TodoList = () => {
  const [todos, setTodos] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  // 加载待办事项
  const loadTodos = async () => {
    try {
      setLoading(true);
      const data = await todoService.getAll();
      setTodos(data);
      setError(null);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  // 创建待办事项
  const handleCreate = async (title, description) => {
    try {
      const newTodo = await todoService.create(title, description);
      setTodos([newTodo, ...todos]);
    } catch (err) {
      alert(err.message);
    }
  };

  // 更新待办事项
  const handleUpdate = async (id, updates) => {
    try {
      const updated = await todoService.update(id, updates);
      setTodos(todos.map(todo => todo.id === id ? updated : todo));
    } catch (err) {
      alert(err.message);
    }
  };

  // 删除待办事项
  const handleDelete = async (id) => {
    try {
      await todoService.delete(id);
      setTodos(todos.filter(todo => todo.id !== id));
    } catch (err) {
      alert(err.message);
    }
  };

  useEffect(() => {
    loadTodos();
  }, []);

  if (loading) return <div className="text-center mt-8">加载中...</div>;
  if (error) return <div className="text-center mt-8 text-red-500">错误：{error}</div>;

  return (
    <div className="max-w-2xl mx-auto mt-8 p-4">
      <h1 className="text-3xl font-bold mb-6">待办事项</h1>
      
      <TodoForm onSubmit={handleCreate} />
      
      <div className="mt-6">
        {todos.length === 0 ? (
          <p className="text-gray-500 text-center">暂无待办事项</p>
        ) : (
          todos.map(todo => (
            <TodoItem
              key={todo.id}
              todo={todo}
              onUpdate={handleUpdate}
              onDelete={handleDelete}
            />
          ))
        )}
      </div>
    </div>
  );
};

export default TodoList;
```

#### 3. 单个待办项组件（`components/TodoItem.js`）

```javascript
import { useState } from 'react';

const TodoItem = ({ todo, onUpdate, onDelete }) => {
  const [isEditing, setIsEditing] = useState(false);
  const [title, setTitle] = useState(todo.title);
  const [description, setDescription] = useState(todo.description);

  const handleToggleComplete = async () => {
    await onUpdate(todo.id, { completed: !todo.completed });
  };

  const handleEdit = () => {
    setIsEditing(true);
  };

  const handleSave = async () => {
    await onUpdate(todo.id, { title, description });
    setIsEditing(false);
  };

  const handleCancel = () => {
    setTitle(todo.title);
    setDescription(todo.description);
    setIsEditing(false);
  };

  const handleDelete = async () => {
    if (window.confirm('确定删除吗？')) {
      await onDelete(todo.id);
    }
  };

  return (
    <div className={`border rounded p-4 mb-4 ${todo.completed ? 'bg-gray-100' : 'bg-white'}`}>
      {isEditing ? (
        // 编辑模式
        <div>
          <input
            type="text"
            value={title}
            onChange={(e) => setTitle(e.target.value)}
            className="border rounded px-2 py-1 w-full mb-2"
          />
          <textarea
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            className="border rounded px-2 py-1 w-full mb-2"
          />
          <div className="flex gap-2">
            <button onClick={handleSave} className="bg-blue-500 text-white px-3 py-1 rounded">
              保存
            </button>
            <button onClick={handleCancel} className="bg-gray-300 px-3 py-1 rounded">
              取消
            </button>
          </div>
        </div>
      ) : (
        // 查看模式
        <div>
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2">
              <input
                type="checkbox"
                checked={todo.completed}
                onChange={handleToggleComplete}
                className="h-5 w-5"
              />
              <span className={`text-lg ${todo.completed ? 'line-through text-gray-500' : ''}`}>
                {todo.title}
              </span>
            </div>
            <div className="flex gap-2">
              <button onClick={handleEdit} className="text-blue-500">
                编辑
              </button>
              <button onClick={handleDelete} className="text-red-500">
                删除
              </button>
            </div>
          </div>
          {todo.description && (
            <p className="text-gray-600 mt-2">{todo.description}</p>
          )}
          <p className="text-xs text-gray-400 mt-2">
            创建时间：{new Date(todo.createdAt).toLocaleString()}
          </p>
        </div>
      )}
    </div>
  );
};

export default TodoItem;
```

#### 4. 创建表单组件（`components/TodoForm.js`）

```javascript
import { useState } from 'react';

const TodoForm = ({ onSubmit }) => {
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!title.trim()) return;
    
    onSubmit(title, description);
    setTitle('');
    setDescription('');
  };

  return (
    <form onSubmit={handleSubmit} className="border rounded p-4 bg-gray-50">
      <h2 className="text-xl font-semibold mb-4">创建新待办</h2>
      <div className="mb-4">
        <input
          type="text"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          placeholder="待办标题"
          className="border rounded px-3 py-2 w-full"
        />
      </div>
      <div className="mb-4">
        <textarea
          value={description}
          onChange={(e) => setDescription(e.target.value)}
          placeholder="描述（可选）"
          className="border rounded px-3 py-2 w-full"
        />
      </div>
      <button type="submit" className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600">
        创建
      </button>
    </form>
  );
};

export default TodoForm;
```

## 核心技巧

### 1. 让 Claude Code 生成"完整可运行"的代码

❌ **不好**：只生成单个组件，不包含样式和 API 调用
✅ **好**：明确说出你要的完整功能

```
请帮我生成一个完整的待办管理页面，包含：
1. 组件代码（List / Item / Form）
2. API 调用层（services/todoService.js）
3. 样式（用 Tailwind CSS）
4. 错误处理和加载状态
```

---

### 2. 用 Tailwind CSS 快速美化界面

Tailwind 是"原子化 CSS"，不用写 CSS 文件，直接在 HTML 里加 class。

**示例**：
```html
<div className="max-w-2xl mx-auto mt-8 p-4">
```

相当于写了：
```css
max-width: 42rem;
margin-left: auto;
margin-right: auto;
margin-top: 2rem;
padding: 1rem;
```

**让 Claude Code 用 Tailwind 写样式**，你不用操心 CSS。

---

### 3. 组件拆分要"合理"

❌ **不好**：所有代码写在一个组件里（TodoList 包含表单、列表、编辑逻辑）
✅ **好**：拆分成多个小组件

- `TodoList`：主组件，管理状态
- `TodoItem`：单个待办项（编辑/删除）
- `TodoForm`：创建待办项的表单

**好处**：
- 每个组件职责单一，易于维护
- 方便复用（比如 `TodoItem` 可以用在搜索结果里）

---

### 4. API 调用层要独立

❌ **不好**：在组件里直接写 `fetch()`
✅ **好**：封装成 `todoService.js`

**好处**：
- 组件代码更简洁
- API 路径变更只需改一个文件
- 方便 mock 数据（测试用）

3. **检查 API 地址**：前端代码里的 `API_BASE` 要指向正确的后端地址
