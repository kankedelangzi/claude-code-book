#### 3. 命令实现（`src/commands/add.js`）

```
请帮我生成 add 命令（src/commands/add.js），需要：
1. 读取 ~/.todo-cli/todos.json
2. 添加新的待办事项（生成唯一 ID）

4. 用 chalk 输出成功信息

请生成完整代码。
```

**Claude Code 会生成类似代码**：

```javascript
const fs = require('fs-extra');
const chalk = require('chalk');
const path = require('path');

const TODOS_FILE = path.join(process.env.HOME, '.todo-cli', 'todos.json');

const addCommand = async (title) => {
  try {
    // 确保目录存在
    await fs.ensureDir(path.dirname(TODOS_FILE));
    
    // 读取现有待办
    let todos = [];
    if (await fs.pathExists(TODOS_FILE)) {
      todos = await fs.readJson(TODOS_FILE);
    }
    
    // 创建新待办
    const newTodo = {
      id: String(todos.length + 1),
      title,
      completed: false,
      createdAt: new Date().toISOString()
    };
    
    todos.push(newTodo);
    
    // 写入文件
    await fs.writeJson(TODOS_FILE, todos, { spaces: 2 });
    
    console.log(chalk.green(`✅ 待办已添加：${title}`));
  } catch (error) {
    console.error(chalk.red(`❌ 添加失败：${error.message}`));
  }
};

module.exports = addCommand;
```

```javascript
const fs = require('fs-extra');
const path = require('path');

const TODOS_FILE = path.join(process.env.HOME, '.todo-cli', 'todos.json');

const storage = {
  // 读取所有待办
  read: async () => {
    try {
      await fs.ensureDir(path.dirname(TODOS_FILE));
      
      if (!await fs.pathExists(TODOS_FILE)) {
        await fs.writeJson(TODOS_FILE, []);
        return [];
      }
      
      return await fs.readJson(TODOS_FILE);
    } catch (error) {
      throw new Error(`读取失败：${error.message}`);
    }
  },
  
  // 写入所有待办
  write: async (todos) => {
    try {
      await fs.ensureDir(path.dirname(TODOS_FILE));
      await fs.writeJson(TODOS_FILE, todos, { spaces: 2 });
    } catch (error) {
      throw new Error(`写入失败：${error.message}`);
    }
  }
};

module.exports = storage;
```

❌ **不好**：用原生 `fs` 模块（代码繁琐）

```javascript
const data = await fs.readJson('file.json');  // 自动解析 JSON
await fs.writeJson('file.json', data);  // 自动格式化
```
