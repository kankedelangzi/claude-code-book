**微服务听起来高大上，但实际落地坑很多。Claude Code 可以帮你避开常见陷阱，快速生成架构设计和代码骨架。**

## ✨ 案例1：电商系统微服务拆分

### 场景

一个电商单体应用越来越庞大，每次部署都要停整个系统，团队决定拆成微服务。

### Claude Code 设计架构

```
设计一个电商系统的微服务架构：

当前单体应用功能：
- 用户注册/登录/权限
- 商品管理（分类、库存、价格）
- 订单创建/查询/取消
- 支付对接（支付宝、微信）
- 物流跟踪

拆分原则：
1. 业务边界清晰
2. 每个服务独立数据库
3. 服务间通过 API 网关通信
4. 核心服务高可用

输出：
1. 服务拆分方案（每个服务的职责和边界）
2. 服务间依赖关系图
3. 数据库设计（每个服务的表）
4. API 网关路由设计
```

### 服务职责划分

| 服务 | 职责 | 数据库表 | 核心接口 |
|------|------|---------|---------|
| User | 用户注册/登录/权限 | users, roles | POST /register, POST /login, GET /me |
| Product | 商品CRUD/库存管理 | products, inventory | GET /products, POST /products, PUT /stock |
| Order | 订单创建/查询/状态 | orders, order_items | POST /orders, GET /orders/:id |
| Payment | 支付创建/回调处理 | payments, transactions | POST /pay, POST /callback |
| Shipping | 物流对接/跟踪 | shipping_records | POST /ship, GET /track/:id |

### 关键代码示例

```javascript
// src/services/authService.js
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

async function register(email, password, name) {
  // 检查邮箱是否已存在
  const existing = await prisma.user.findUnique({ where: { email } });
  if (existing) {
    throw new Error('邮箱已被注册');
  }

  // 哈希密码
  const hashedPassword = await bcrypt.hash(password, 10);

  // 创建用户
  const user = await prisma.user.create({
    data: { email, password: hashedPassword, name }
  });

  // 生成 JWT
  const token = jwt.sign(
    { userId: user.id, email: user.email },
    process.env.JWT_SECRET,
    { expiresIn: '7d' }
  );

  return { user: { id: user.id, email: user.email, name: user.name }, token };
}

async function login(email, password) {
  const user = await prisma.user.findUnique({ where: { email } });
  if (!user) {
    throw new Error('用户不存在');
  }

  const valid = await bcrypt.compare(password, user.password);
  if (!valid) {
    throw new Error('密码错误');
  }

  const token = jwt.sign(
    { userId: user.id, email: user.email },
    process.env.JWT_SECRET,
    { expiresIn: '7d' }
  );

  return { user: { id: user.id, email: user.email, name: user.name }, token };
}

module.exports = { register, login };
```

```javascript
// src/middleware/auth.js
const jwt = require('jsonwebtoken');

function authMiddleware(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: '未登录' });
  }

  const token = authHeader.slice(7);
  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET);
    req.user = payload;
    next();
  } catch (err) {
    return res.status(401).json({ error: 'token 无效' });
  }
}

module.exports = authMiddleware;
```

## ✨ 案例3：服务间通信——订单服务调用支付服务

### 场景

订单创建后需要调用支付服务创建支付单。如何实现服务间通信？

### 同步通信方案（HTTP）

```javascript
// order-service/src/services/paymentClient.js
const axios = require('axios');

const PAYMENT_SERVICE_URL = process.env.PAYMENT_SERVICE_URL || 'http://payment-service:3004';

async function createPayment(orderId, amount, method) {
  try {
    const response = await axios.post(
      `${PAYMENT_SERVICE_URL}/api/payments`,
      { orderId, amount, method },
      { timeout: 5000 }
    );
    return response.data;
  } catch (err) {
    if (err.code === 'ECONNABORTED') {
      throw new Error('支付服务超时');
    }
    throw new Error('支付服务异常: ' + err.message);
  }
}

module.exports = { createPayment };
```

### 异步通信方案（消息队列）

```javascript
// order-service/src/services/orderService.js
const { PrismaClient } = require('@prisma/client');
const amqp = require('amqplib');
const prisma = new PrismaClient();

let channel;

async function connectMQ() {
  const connection = await amqp.connect(process.env.RABBITMQ_URL);
  channel = await connection.createChannel();
  await channel.assertQueue('payment.created');
  await channel.assertQueue('order.created');
}

async function createOrder(userId, items) {
  // 计算总价
  const total = items.reduce((sum, item) => sum + item.price * item.quantity, 0);

  // 创建订单
  const order = await prisma.order.create({
    data: {
      userId,
      status: 'pending',
      total,
      items: {
        create: items.map(item => ({
          productId: item.productId,
          quantity: item.quantity,
          price: item.price
        }))
      }
    }
  });

  // 发消息到队列，让支付服务处理
  await channel.sendToQueue(
    'order.created',
    Buffer.from(JSON.stringify({
      orderId: order.id,
      userId,
      total,
      createdAt: order.createdAt
    }))
  );

  return order;
}

module.exports = { connectMQ, createOrder };
```

```javascript
// payment-service/src/services/orderConsumer.js
const amqp = require('amqplib');
const { createPayment } = require('./paymentService');

async function consumeOrders() {
  const connection = await amqp.connect(process.env.RABBITMQ_URL);
  const channel = await connection.createChannel();
  await channel.assertQueue('order.created');

  channel.consume('order.created', async (msg) => {
    const order = JSON.parse(msg.content.toString());
    console.log('收到新订单:', order.orderId);

    try {
      // 创建支付单
      const payment = await createPayment(order.orderId, order.total, 'pending');
      
      // 通知订单服务支付单已创建
      channel.sendToQueue(
        'payment.created',
        Buffer.from(JSON.stringify({
          orderId: order.orderId,
          paymentId: payment.id,
          status: 'ready'
        }))
      );

      channel.ack(msg);
    } catch (err) {
      console.error('支付创建失败:', err);
      // 不 ack，让消息重新入队
      channel.nack(msg, false, true);
    }
  });
}

module.exports = { consumeOrders };
```

### 两种方案对比

| 方案 | 适用场景 | 优点 | 缺点 |
|------|---------|------|------|
| HTTP 同步 | 实时性要求高、逻辑简单 | 实现简单、调试方便 | 服务依赖强、超时风险 |
| 消息队列异步 | 高并发、需要解耦 | 解耦、削峰填谷 | 复杂度高、需要处理重复消费 |

## ✨ 案例4：Docker Compose 本地开发环境

### 场景

微服务本地调试困难，每个服务要单独启动。用 Docker Compose 统一管理。

### Claude Code 生成配置

```
生成 docker-compose.yml，包含以下服务：
- user-service (端口 3001)
- product-service (端口 3002)
- order-service (端口 3003)
- payment-service (端口 3004)
- api-gateway (端口 8080)
- PostgreSQL（每个服务独立数据库）
- RabbitMQ（消息队列）

所有服务共享同一个网络，可以互相访问。
```

### 生成的配置

```yaml
# docker-compose.yml
version: '3.8'

services:
  # PostgreSQL 主实例
  postgres:
    image: postgres:15
    environment:
      POSTGRES_PASSWORD: postgres
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data

  # RabbitMQ
  rabbitmq:
    image: rabbitmq:3-management
    ports:
      - "5672:5672"
      - "15672:15672"  # 管理界面
    volumes:
      - rabbitmq_data:/var/lib/rabbitmq

  # User Service
  user-service:
    build: ./user-service
    ports:
      - "3001:3001"
    environment:
      DATABASE_URL: postgresql://postgres:postgres@postgres:5432/user_db
      JWT_SECRET: dev_secret
      PORT: 3001
    depends_on:
      - postgres

  # Product Service
  product-service:
    build: ./product-service
    ports:
      - "3002:3002"
    environment:
      DATABASE_URL: postgresql://postgres:postgres@postgres:5432/product_db
      PORT: 3002
    depends_on:
      - postgres

  # Order Service
  order-service:
    build: ./order-service
    ports:
      - "3003:3003"
    environment:
      DATABASE_URL: postgresql://postgres:postgres@postgres:5432/order_db
      RABBITMQ_URL: amqp://guest:guest@rabbitmq:5672
      PAYMENT_SERVICE_URL: http://payment-service:3004
      PORT: 3003
    depends_on:
      - postgres
      - rabbitmq
      - payment-service

  # Payment Service
  payment-service:
    build: ./payment-service
    ports:
      - "3004:3004"
    environment:
      DATABASE_URL: postgresql://postgres:postgres@postgres:5432/payment_db
      RABBITMQ_URL: amqp://guest:guest@rabbitmq:5672
      PORT: 3004
    depends_on:
      - postgres
      - rabbitmq

  # API Gateway
  api-gateway:
    image: nginx:alpine
    ports:
      - "8080:80"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
    depends_on:
      - user-service
      - product-service
      - order-service
      - payment-service

volumes:
  postgres_data:
  rabbitmq_data:

networks:
  default:
    name: microservices-net
```

## ⚠️ 微服务落地注意事项

| 陷阱 | 说明 | Claude Code 如何帮你避开 |
|------|------|-------------------------|
| **拆分过度** | 什么都拆成服务，复杂度爆炸 | Claude 会建议合理的拆分边界 |
| **服务间强耦合** | A 挂了 B 也挂 | Claude 会设计异步通信和解耦方案 |
| **数据一致性难保证** | 跨服务事务复杂 | Claude 会建议 Saga 模式或最终一致性 |
| **调试困难** | 本地跑不全服务 | Claude 生成 Docker Compose 一键启动 |
| **监控缺失** | 服务挂了没人知道 | Claude 会建议日志和告警方案 |
