## 服务间通信方式

| 场景 | 通信方式 | 理由 |
|---|---|---|
| 创建待办时验证用户 | HTTP 调用（User Service） | 同步调用，立即返回结果 |
| 待办到期提醒 | Redis Pub/Sub | 异步处理，解耦 |

## 数据库设计

### 用户数据库（user_db）
```javascript
const UserSchema = new mongoose.Schema({
  name: String,
  email: { type: String, unique: true },
  password: String (hashed)
});
```

### 待办数据库（todo_db）
```javascript
const TodoSchema = new mongoose.Schema({
  title: String,
  description: String,
  userId: ObjectId (ref: User),  // 注意：跨库无法建外键
  completed: Boolean,
  dueDate: Date
});
```

**注意**：微服务架构中，每个服务有独立数据库，无法使用数据库外键，需要**应用层保证一致性**。

## 分布式事务方案

### 场景：创建待办事项（需要验证用户是否存在）

**方案：SAGA 模式**
1. Todo Service 发送 "创建待办" 消息到消息队列
2. User Service 消费消息，验证用户是否存在
3. 如果存在，Todo Service 创建待办；否则回滚

- 每个微服务独立容器
- 共用 Redis（消息队列）
- 共用 MongoDB（但不同数据库）

### 方案 2：Kubernetes（生产环境）
- 每个微服务独立 Pod
- 用 Service 暴露服务
- 用 Ingress 路由外部请求

```javascript
const User = require('../models/User');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

const userController = {
  // 创建用户
  create: async (req, res, next) => {
    try {
      const { name, email, password } = req.body;
      
      // 检查用户是否已存在
      const existing = await User.findOne({ email });
      if (existing) {
        return res.status(400).json({ error: 'Email 已存在' });
      }
      
      // 加密密码
      const hashedPassword = await bcrypt.hash(password, 10);
      
      const user = new User({
        name,
        email,
        password: hashedPassword
      });
      
      await user.save();
      
      res.status(201).json({
        id: user._id,
        name: user.name,
        email: user.email
      });
    } catch (error) {
      next(error);
    }
  },
  
  // 登录
  login: async (req, res, next) => {
    try {
      const { email, password } = req.body;
      
      const user = await User.findOne({ email });
      if (!user) {
        return res.status(401).json({ error: '用户不存在' });
      }
      
      // 验证密码
      const isValid = await bcrypt.compare(password, user.password);
      if (!isValid) {
        return res.status(401).json({ error: '密码错误' });
      }
      
      // 生成 JWT
      const token = jwt.sign(
        { userId: user._id },
        process.env.JWT_SECRET,
        { expiresIn: '7d' }
      );
      
      res.json({ token });
    } catch (error) {
      next(error);
    }
  },
  
  // 获取用户信息
  getById: async (req, res, next) => {
    try {
      const user = await User.findById(req.params.id).select('-password');
      if (!user) {
        return res.status(404).json({ error: '用户不存在' });
      }
      res.json(user);
    } catch (error) {
      next(error);
    }
  }
};

module.exports = userController;
```

**Claude Code 会生成类似代码**：

**`todo-service/src/services/todoService.js`**：

```javascript
const Todo = require('../models/Todo');
const redis = require('../config/redis');

const todoService = {
  // 创建待办（SAGA 模式）
  createWithSaga: async (todoData) => {
    // 步骤1：发送 "验证用户" 消息到消息队列
    const sagaId = generateSagaId();
    
    await redis.publish('saga-events', JSON.stringify({
      sagaId,
      type: 'VALIDATE_USER',
      todoData
    }));
    
    // 等待 User Service 的响应（用 Redis 订阅）
    return new Promise((resolve, reject) => {
      const subscriber = redis.duplicate();
      subscriber.subscribe('saga-responses', (message) => {
        const response = JSON.parse(message);
        
        if (response.sagaId === sagaId) {
          if (response.type === 'USER_VALIDATED') {
            // 用户验证通过，创建待办
            const todo = new Todo(todoData);
            todo.save()
              .then(todo => resolve(todo))
              .catch(reject);
          } else if (response.type === 'USER_VALIDATION_FAILED') {
            // 用户验证失败，回滚
            reject(new Error('用户不存在'));
          }
          
          subscriber.unsubscribe('saga-responses');
        }
      });
    });
  }
};

// 生成唯一 SAGA ID
const generateSagaId = () => {
  return `saga-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
};

module.exports = todoService;
```

**`user-service/src/saga/sagaConsumer.js`**：

```javascript
const redis = require('../config/redis');
const User = require('../models/User');

const startSagaConsumer = async () => {
  console.log('📦 SAGA 消费者启动...');
  
  const subscriber = redis.duplicate();
  await subscriber.connect();
  
  await subscriber.subscribe('saga-events', async (message) => {
    const event = JSON.parse(message);
    
    if (event.type === 'VALIDATE_USER') {
      const { sagaId, todoData } = event;
      
      try {
        // 验证用户是否存在
        const user = await User.findById(todoData.userId);
        
        if (user) {
          // 验证通过，发送响应
          await redis.publish('saga-responses', JSON.stringify({
            sagaId,
            type: 'USER_VALIDATED'
          }));
        } else {
          // 验证失败，发送响应
          await redis.publish('saga-responses', JSON.stringify({
            sagaId,
            type: 'USER_VALIDATION_FAILED'
          }));
        }
      } catch (error) {
        console.error('SAGA 处理失败：', error);
      }
    }
  });
};

module.exports = startSagaConsumer;
```
