### 第二步：让 Claude Code 生成数据读取脚本（5分钟）

对 Claude Code 说：

```
请帮我生成 CSV 数据读取脚本，需求：
1. 用 csv-parser 读取 CSV 文件
2. 处理每一行数据（转换为 JSON）
3. 错误处理（文件不存在、格式错误）
4. 返回 JSON 数组

请生成完整代码（scripts/process-sales.js）。
```

**Claude Code 会生成类似代码**：

#### `scripts/process-sales.js`

```javascript
#!/usr/bin/env node

const fs = require('fs');
const csv = require('csv-parser');
const path = require('path');

const DATA_FILE = process.argv[2] || path.join(__dirname, '../data/sales.csv');

const processSalesData = async () => {
  try {
    const results = [];
    
    // 用 stream 读取 CSV（适合大文件）
    await new Promise((resolve, reject) => {
      fs.createReadStream(DATA_FILE)
        .pipe(csv())
        .on('data', (data) => {
          // 数据清洗：转换类型
          results.push({
            date: data.date,
            product: data.product,
            quantity: parseInt(data.quantity),
            price: parseFloat(data.price),
            region: data.region
          });
        })
        .on('end', () => {
          console.log(`✅ 读取完成，共 ${results.length} 条数据`);
          resolve(results);
        })
        .on('error', (error) => {
          reject(error);
        });
    });
    
    return results;
  } catch (error) {
    console.error('❌ 读取失败：', error.message);
    throw error;
  }
};

// 如果直接运行脚本（node scripts/process-sales.js）
if (require.main === module) {
  processSalesData()
    .then(data => {
      console.log(JSON.stringify(data, null, 2));
      process.exit(0);
    })
    .catch(error => {
      console.error(error);
      process.exit(1);
    });
}

module.exports = { processSalesData };
```

2. 按地区统计销售总额
3. 找出销售额最高的产品

**Claude Code 会生成类似代码**：

#### `scripts/analyze-sales.js`

```javascript
const { processSalesData } = require('./process-sales');

const analyzeSalesData = async () => {
  try {
    const salesData = await processSalesData();
    
    // 按产品统计
    const salesByProduct = {};
    salesData.forEach(row => {
      const key = row.product;
      if (!salesByProduct[key]) {
        salesByProduct[key] = {
          totalQuantity: 0,
          totalRevenue: 0
        };
      }
      salesByProduct[key].totalQuantity += row.quantity;
      salesByProduct[key].totalRevenue += row.quantity * row.price;
    });
    
    // 按地区统计
    const salesByRegion = {};
    salesData.forEach(row => {
      const key = row.region;
      if (!salesByRegion[key]) {
        salesByRegion[key] = {
          totalQuantity: 0,
          totalRevenue: 0
        };
      }
      salesByRegion[key].totalQuantity += row.quantity;
      salesByRegion[key].totalRevenue += row.quantity * row.price;
    });
    
    // 找出销售额最高的产品
    let topProduct = null;
    let topProductRevenue = 0;
    for (const [product, data] of Object.entries(salesByProduct)) {
      if (data.totalRevenue > topProductRevenue) {
        topProduct = product;
        topProductRevenue = data.totalRevenue;
      }
    }
    
    // 找出销售额最高的地区
    let topRegion = null;
    let topRegionRevenue = 0;
    for (const [region, data] of Object.entries(salesByRegion)) {
      if (data.totalRevenue > topRegionRevenue) {
        topRegion = region;
        topRegionRevenue = data.totalRevenue;
      }
    }
    
    const analysis = {
      salesByProduct,
      salesByRegion,
      topProduct: {
        name: topProduct,
        revenue: topProductRevenue
      },
      topRegion: {
        name: topRegion,
        revenue: topRegionRevenue
      }
    };
    
    console.log('✅ 数据分析完成');
    return analysis;
  } catch (error) {
    console.error('❌ 分析失败：', error.message);
    throw error;
  }
};

// 如果直接运行脚本
if (require.main === module) {
  analyzeSalesData()
    .then(analysis => {
      console.log(JSON.stringify(analysis, null, 2));
      process.exit(0);
    })
    .catch(error => {
      console.error(error);
      process.exit(1);
    });
}

module.exports = { analyzeSalesData };
```

#### `scripts/export-report.js`

```javascript
const fs = require('fs-extra');
const path = require('path');
const ExcelJS = require('exceljs');
const { analyzeSalesData } = require('./analyze-sales');

const exportReport = async () => {
  try {
    const analysis = await analyzeSalesData();
    const reportsDir = path.join(__dirname, '../reports');
    
    // 确保目录存在
    await fs.ensureDir(reportsDir);
    
    // 1. 导出为 JSON
    const jsonPath = path.join(reportsDir, 'sales-report.json');
    await fs.writeJson(jsonPath, analysis, { spaces: 2 });
    console.log(`✅ JSON 报表已导出：${jsonPath}`);
    
    // 2. 导出为 CSV
    const csvPath = path.join(reportsDir, 'sales-report.csv');
    const csvLines = [
      'type,key,totalQuantity,totalRevenue'
    ];
    
    // 按产品统计
    for (const [product, data] of Object.entries(analysis.salesByProduct)) {
      csvLines.push(`product,${product},${data.totalQuantity},${data.totalRevenue}`);
    }
    
    // 按地区统计
    for (const [region, data] of Object.entries(analysis.salesByRegion)) {
      csvLines.push(`region,${region},${data.totalQuantity},${data.totalRevenue}`);
    }
    
    await fs.writeFile(csvPath, csvLines.join('\n'));
    console.log(`✅ CSV 报表已导出：${csvPath}`);
    
    // 3. 导出为 Excel
    const excelPath = path.join(reportsDir, 'sales-report.xlsx');
    const workbook = new ExcelJS.Workbook();
    
    // 工作表：按产品统计
    const productSheet = workbook.addWorksheet('按产品统计');
    productSheet.columns = [
      { header: '产品', key: 'product' },
      { header: '总销量', key: 'totalQuantity' },
      { header: '总销售额', key: 'totalRevenue' }
    ];
    for (const [product, data] of Object.entries(analysis.salesByProduct)) {
      productSheet.addRow({ product, ...data });
    }
    
    // 工作表：按地区统计
    const regionSheet = workbook.addWorksheet('按地区统计');
    regionSheet.columns = [
      { header: '地区', key: 'region' },
      { header: '总销量', key: 'totalQuantity' },
      { header: '总销售额', key: 'totalRevenue' }
    ];
    for (const [region, data] of Object.entries(analysis.salesByRegion)) {
      regionSheet.addRow({ region, ...data });
    }
    
    // 工作表：Top 统计
    const topSheet = workbook.addWorksheet('Top 统计');
    topSheet.columns = [
      { header: '类型', key: 'type' },
      { header: '名称', key: 'name' },
      { header: '销售额', key: 'revenue' }
    ];
    topSheet.addRow({ type: '产品', name: analysis.topProduct.name, revenue: analysis.topProduct.revenue });
    topSheet.addRow({ type: '地区', name: analysis.topRegion.name, revenue: analysis.topRegion.revenue });
    
    await workbook.xlsx.writeFile(excelPath);
    console.log(`✅ Excel 报表已导出：${excelPath}`);
    
    return {
      json: jsonPath,
      csv: csvPath,
      excel: excelPath
    };
  } catch (error) {
    console.error('❌ 导出失败：', error.message);
    throw error;
  }
};

// 如果直接运行脚本
if (require.main === module) {
  exportReport()
    .then(result => {
      console.log('✅ 报表导出完成');
      console.log(result);
      process.exit(0);
    })
    .catch(error => {
      console.error(error);
      process.exit(1);
    });
}

module.exports = { exportReport };
```

### 第五步：让 Claude Code 优化性能（5分钟）

如果数据量很大（百万行），让 Claude Code 优化：

```
我的 CSV 文件有 100 万行，处理很慢。请帮我优化：
1. 用 stream 分批处理（每次处理 1000 行）
2. 用 worker_threads 并行处理
3. 优化内存占用（不一次性加载所有数据）

请生成优化后的代码。
```

**Claude Code 会生成优化后的代码**：

```javascript
const fs = require('fs');
const csv = require('csv-parser');
const { Worker, isMainThread, parentPort, workerData } = require('worker_threads');

const BATCH_SIZE = 1000;

// 主线程：分批读取 CSV
const processLargeCSV = async (filePath) => {
  let batch = [];
  const workers = [];
  
  await new Promise((resolve, reject) => {
    fs.createReadStream(filePath)
      .pipe(csv())
      .on('data', (data) => {
        batch.push(data);
        
        // 达到批次大小，交给 worker 处理
        if (batch.length >= BATCH_SIZE) {
          const worker = new Worker(__filename, { workerData: batch });
          workers.push(worker);
          batch = [];
        }
      })
      .on('end', () => {
        // 处理最后一批
        if (batch.length > 0) {
          const worker = new Worker(__filename, { workerData: batch });
          workers.push(worker);
        }
        
        // 等待所有 worker 完成
        Promise.all(workers.map(w => new Promise(resolve => w.on('exit', resolve))))
          .then(() => resolve());
      })
      .on('error', reject);
  });
};

// Worker 线程：处理批次数据
if (!isMainThread) {
  const batch = workerData;
  // 处理数据...
  parentPort.postMessage({ success: true, processed: batch.length });
}

// 如果直接运行脚本
if (require.main === module) {
  processLargeCSV(process.argv[2])
    .then(() => {
      console.log('✅ 大数据处理完成');
      process.exit(0);
    })
    .catch(error => {
      console.error(error);
      process.exit(1);
    });
}
```
