```typescript
export async function getMongoClient() {
  if (!mongoClient) {
    mongoClient = new MongoClient(process.env.MONGO_URL!);

    console.log('✅ MongoDB connected');
  }

// Redis 客户端
let redisClient: ReturnType<typeof createClient> | null = null;

export async function getRedisClient() {
  if (!redisClient) {
    redisClient = createClient({ url: process.env.REDIS_URL });
```

**Claude Code 生成代码**（节选）：

```typescript
// backend/src/services/DataExtractionService.ts

import { getMySQLConnection } from '../config/database';
import { getMongoClient } from '../config/database';
import { getRedisClient } from '../config/database';

export interface SalesData {
  orderId: string;
  userId: string;
  productId: string;
  amount: number;
  createdAt: Date;
}

export interface UserEvent {
  userId: string;
  eventType: 'click' | 'view' | 'purchase';
  page: string;
  timestamp: Date;
}

export class DataExtractionService {
  // 从 MySQL 提取销售数据
  async getSalesData(options: {
    startDate?: Date;
    endDate?: Date;
    groupBy?: 'product' | 'user' | 'day';
  }): Promise<SalesData[]> {
    const cacheKey = `sales:${JSON.stringify(options)}`;
    const redis = await getRedisClient();

    // 尝试从缓存读取
    const cached = await redis.get(cacheKey);
    if (cached) {
      console.log('📦 Cache hit:', cacheKey);
      return JSON.parse(cached);
    }

    // 从 MySQL 查询
    const mysql = await getMySQLConnection();
    let query = 'SELECT * FROM orders WHERE 1=1';
    const params: any[] = [];

    if (options.startDate) {
      query += ' AND created_at >= ?';
      params.push(options.startDate);
    }
    if (options.endDate) {
      query += ' AND created_at <= ?';
      params.push(options.endDate);
    }

    // 分组逻辑
    if (options.groupBy === 'product') {
      query = `
        SELECT product_id, SUM(amount) as total_amount, COUNT(*) as order_count
        FROM orders
        WHERE 1=1
        ${options.startDate ? 'AND created_at >= ?' : ''}
        ${options.endDate ? 'AND created_at <= ?' : ''}
        GROUP BY product_id
        ORDER BY total_amount DESC
      `;
    }

    const [rows] = await mysql.execute(query, params);
    const data = rows as SalesData[];

    // 写入缓存（TTL 5 分钟）
    await redis.setEx(cacheKey, 300, JSON.stringify(data));
    console.log('💾 Cache set:', cacheKey);

    return data;
  }

  // 从 MongoDB 提取用户行为数据
  async getUserEvents(options: {
    eventType?: string;
    startDate?: Date;
    endDate?: Date;
    limit?: number;
  }): Promise<UserEvent[]> {
    const db = await getMongoClient();
    const collection = db.collection<UserEvent>('events');

    const query: any = {};
    if (options.eventType) {
      query.event_type = options.eventType;
    }
    if (options.startDate || options.endDate) {
      query.timestamp = {};
      if (options.startDate) query.timestamp.$gte = options.startDate;
      if (options.endDate) query.timestamp.$lte = options.endDate;
    }

    const events = await collection
      .find(query)
      .limit(options.limit || 1000)
      .toArray();

    return events;
  }

  // 从 Redis 读取实时热销商品
  async getHotProducts(limit: number = 10): Promise<Array<{ productId: string; sales: number }>> {
    const redis = await getRedisClient();
    const results = await redis.zRangeWithScores('hot:products', 0, limit - 1, { REV: true });
    
    return results.map((item) => ({
      productId: item.value,
      sales: item.score,
    }));
  }
}
```

- ✅ 自动添加缓存逻辑（Redis，TTL 5 分钟）
- ✅ 支持灵活的时间范围过滤
- ✅ 支持按商品/用户/日期分组

```tsx
import React, { useEffect, useState } from 'react';
import { LineChartOutlined } from '@ant-design/icons';
import { Card, DatePicker, Spin } from 'antd';
import ReactECharts from 'echarts-for-react';
import { DataExtractionService } from '../services/DataExtractionService';

const { RangePicker } = DatePicker;

export const SalesTrendChart: React.FC = () => {
  const [data, setData] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [dateRange, setDateRange] = useState<[moment.Moment, moment.Moment] | null>(null);

  useEffect(() => {
    fetchData();
  }, [dateRange]);

  const fetchData = async () => {
    setLoading(true);
    try {
      const service = new DataExtractionService();
      const salesData = await service.getSalesData({
        startDate: dateRange ? dateRange[0].toDate() : undefined,
        endDate: dateRange ? dateRange[1].toDate() : undefined,

      });

      // 转换为 ECharts 格式
      const chartData = salesData.map((item) => ({
        date: moment(item.createdAt).format('MM-DD'),
        amount: item.amount,
      }));

      setData(chartData);
    } catch (error) {
      console.error('Failed to fetch sales data:', error);
    } finally {
      setLoading(false);
    }
  };

  const getChartOption = () => ({
    title: {
      text: '销售额趋势',
      left: 'center',
    },
    tooltip: {
      trigger: 'axis',
      formatter: (params: any) => {
        const data = params[0];
        return `${data.name}<br/>销售额: ¥${data.value.toLocaleString()}`;
      },
    },
    xAxis: {
      type: 'category',
      data: data.map((item) => item.date),
    },
    yAxis: {
      type: 'value',
      axisLabel: {
        formatter: '¥{value}',
      },
    },
    series: [
      {
        data: data.map((item) => item.amount),
        type: 'line',
        smooth: true,
        areaStyle: {
          opacity: 0.3,
        },
      },
    ],
  });

  return (
    <Card
      title={
        <>
          <LineChartOutlined /> 销售额趋势
        </>
      }
      extra={
        <RangePicker
          onChange={(dates) => setDateRange(dates)}

            { label: '最近 7 天', value: [moment().subtract(7, 'days'), moment()] },

  );
```

```typescript
// prediction/LSTMModel.ts

import * as tf from '@tensorflow/tfjs-node';
import { DataExtractionService } from '../backend/src/services/DataExtractionService';

export class LSTMModel {
  private model: tf.Sequential | null = null;
  private readonly SEQUENCE_LENGTH = 30; // 使用过去 30 天预测未来
  private readonly PREDICTION_DAYS = 7;

  // 创建 LSTM 模型
  private createModel() {
    const model = tf.sequential();

    // LSTM 层
    model.add(
      tf.layers.lstm({
        units: 50,
        returnSequences: true,
        inputShape: [this.SEQUENCE_LENGTH, 1],
      })
    );

    model.add(
      tf.layers.lstm({
        units: 50,
        returnSequences: false,
      })
    );

    // 全连接层
    model.add(tf.layers.dense({ units: this.PREDICTION_DAYS }));

    // 编译模型
    model.compile({
      optimizer: tf.train.adam(0.001),
      loss: 'meanSquaredError',
    });

    return model;
  }

  // 训练模型
  async train(historicalData: number[]) {
    console.log('🚀 Starting LSTM training...');
    
    this.model = this.createModel();

    // 准备训练数据
    const { xs, ys } = this.prepareTrainingData(historicalData);

    // 转换张量
    const xsTensor = tf.tensor3d(xs, [xs.length, this.SEQUENCE_LENGTH, 1]);
    const ysTensor = tf.tensor2d(ys, [ys.length, this.PREDICTION_DAYS]);

    // 训练
    await this.model.fit(xsTensor, ysTensor, {
      epochs: 100,
      batchSize: 32,
      validationSplit: 0.2,
      callbacks: {
        onEpochEnd: (epoch, logs) => {
          console.log(`Epoch ${epoch + 1}: loss = ${logs?.loss}`);
        },
      },
    });

    console.log('✅ Training complete');

    // 保存模型
    await this.model.save('file://./prediction/models/lstm');
  }

  // 预测
  async predict(latestData: number[]): Promise<number[]> {
    if (!this.model) {
      // 尝试加载已训练的模型
      try {
        this.model = await tf.loadLayersModel('file://./prediction/models/lstm/model.json');
        console.log('✅ Model loaded from disk');
      } catch (error) {
        throw new Error('Model not trained. Please train first.');
      }
    }

    // 准备输入数据
    const input = latestData.slice(-this.SEQUENCE_LENGTH);
    const inputTensor = tf.tensor3d([input], [1, this.SEQUENCE_LENGTH, 1]);

    // 预测
    const prediction = this.model.predict(inputTensor) as tf.Tensor;
    const result = await prediction.array();

    return result[0];
  }

  // 准备训练数据
  private prepareTrainingData(data: number[]) {
    const xs: number[][][] = [];
    const ys: number[][] = [];

    for (let i = 0; i < data.length - this.SEQUENCE_LENGTH - this.PREDICTION_DAYS; i++) {
      const x = data.slice(i, i + this.SEQUENCE_LENGTH);
      const y = data.slice(i + this.SEQUENCE_LENGTH, i + this.SEQUENCE_LENGTH + this.PREDICTION_DAYS);
      xs.push(x.map((val) => [val]));
      ys.push(y);
    }

    return { xs, ys };
  }
}
```

**Claude Code 生成代码**（节选）：

```typescript
// reports/HtmlReportGenerator.ts

import Handlebars from 'handlebars';
import fs from 'fs';
import path from 'path';
import { DataExtractionService } from '../backend/src/services/DataExtractionService';
import { PredictionService } from '../prediction/PredictionService';

export class HtmlReportGenerator {
  private template: Handlebars.TemplateDelegate;

  constructor() {
    // 加载模板
    const templatePath = path.join(__dirname, 'templates', 'report.hbs');
    const templateContent = fs.readFileSync(templatePath, 'utf-8');
    this.template = Handlebars.compile(templateContent);
  }

  // 生成报告
  async generate(options: {
    startDate: Date;
    endDate: Date;
    title?: string;
  }): Promise<string> {
    console.log('📝 Generating HTML report...');

    // 1. 提取数据
    const dataService = new DataExtractionService();
    const salesData = await dataService.getSalesData({
      startDate: options.startDate,
      endDate: options.endDate,
      groupBy: 'day',
    });

    // 2. 生成预测
    const predictionService = new PredictionService();
    const prediction = await predictionService.predictNextWeek();

    // 3. 计算关键指标
    const totalSales = salesData.reduce((sum, item) => sum + item.amount, 0);
    const averageSales = totalSales / salesData.length;
    const growthRate = this.calculateGrowthRate(salesData);

    // 4. 渲染模板
    const html = this.template({
      title: options.title || '电商数据分析报告',
      startDate: options.startDate.toLocaleDateString(),
      endDate: options.endDate.toLocaleDateString(),
      totalSales: totalSales.toLocaleString(),
      averageSales: averageSales.toLocaleString(),
      growthRate: (growthRate * 100).toFixed(2) + '%',
      salesData: salesData.slice(0, 10), // 只显示前 10 条
      prediction: prediction,
      generatedAt: new Date().toLocaleString(),
    });

    // 5. 保存文件
    const outputPath = path.join(
      __dirname,
      'output',
      `report_${Date.now()}.html`
    );
    fs.writeFileSync(outputPath, html);

    console.log('✅ Report generated:', outputPath);
    return outputPath;
  }

  // 计算增长率
  private calculateGrowthRate(data: any[]): number {
    if (data.length < 2) return 0;
    
    const firstHalf = data.slice(0, Math.floor(data.length / 2));
    const secondHalf = data.slice(Math.floor(data.length / 2));

    const firstAvg = firstHalf.reduce((sum, item) => sum + item.amount, 0) / firstHalf.length;
    const secondAvg = secondHalf.reduce((sum, item) => sum + item.amount, 0) / secondHalf.length;

    return (secondAvg - firstAvg) / firstAvg;
  }
}
```
