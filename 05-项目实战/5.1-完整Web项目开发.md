#### `frontend/src/services/todoService.ts`

```typescript
import axios, { AxiosInstance } from 'axios';

const API_BASE = process.env.REACT_APP_API_URL || 'http://localhost:3000/api';

class TodoService {
  private api: AxiosInstance;

  constructor() {
    this.api = axios.create({
      baseURL: API_BASE,
      headers: {
        'Content-Type': 'application/json'
      }
    });

    // 请求拦截器：添加 JWT token
    this.api.interceptors.request.use(config => {
      const token = localStorage.getItem('token');
      if (token) {
        config.headers.Authorization = `Bearer ${token}`;
      }
      return config;
    });
  }

  // 获取所有待办事项
  async getAll(): Promise<Todo[]> {
    const response = await this.api.get('/todos');
    return response.data;
  }

  // 创建待办事项
  async create(todo: Partial<Todo>): Promise<Todo> {
    const response = await this.api.post('/todos', todo);
    return response.data;
  }

  // 更新待办事项
  async update(id: string, updates: Partial<Todo>): Promise<Todo> {
    const response = await this.api.put(`/todos/${id}`, updates);
    return response.data;
  }

  // 删除待办事项
  async delete(id: string): Promise<void> {
    await this.api.delete(`/todos/${id}`);
  }
}

export default new TodoService();
```

#### `frontend/src/pages/TodoList.tsx`

```typescript
import { useEffect, useState } from 'react';
import TodoService from '../services/todoService';
import TodoItem from '../components/TodoItem';

const TodoList = () => {
  const [todos, setTodos] = useState<Todo[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    loadTodos();
  }, []);

  const loadTodos = async () => {
    try {
      setLoading(true);
      const data = await TodoService.getAll();
      setTodos(data);
      setError(null);
    } catch (err: any) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  const handleDelete = async (id: string) => {
    try {
      await TodoService.delete(id);
      setTodos(todos.filter(todo => todo._id !== id));
    } catch (err: any) {
      setError(err.message);
    }
  };

  if (loading) return <div className="text-center mt-8">加载中...</div>;
  if (error) return <div className="text-center mt-8 text-red-500">{error}</div>;

  return (
    <div className="max-w-2xl mx-auto mt-8 p-4">
      <h1 className="text-3xl font-bold mb-6">待办事项</h1>
      
      <TodoForm onSuccess={loadTodos} />
      
      <div className="mt-6">
        {todos.length === 0 ? (
          <p className="text-gray-500 text-center">暂无待办事项</p>
        ) : (
          todos.map(todo => (
            <TodoItem
              key={todo._id}
              todo={todo}
              onDelete={handleDelete}
            />
          ))
        )}
      </div>
    </div>
  );
};

export default TodoList;
```

#### `backend/src/controllers/todoController.js`

```javascript
const Todo = require('../models/Todo');

const todoController = {
  // 获取所有待办事项
  getAll: async (req, res, next) => {
    try {
      const todos = await Todo.find({ userId: req.user.id });
      res.json(todos);
    } catch (error) {
      next(error);
    }
  },

  // 创建待办事项
  create: async (req, res, next) => {
    try {
      const { title, description } = req.body;
      
      const todo = new Todo({
        title,
        description,
        userId: req.user.id
      });
      
      await todo.save();
      res.status(201).json(todo);
    } catch (error) {
      next(error);
    }
  },

  // 更新待办事项
  update: async (req, res, next) => {
    try {
      const { title, description, completed } = req.body;
      
      const todo = await Todo.findOneAndUpdate(
        { _id: req.params.id, userId: req.user.id },
        { title, description, completed },
        { new: true, runValidators: true }
      );
      
      if (!todo) {
        return res.status(404).json({ error: 'Todo not found' });
      }
      
      res.json(todo);
    } catch (error) {
      next(error);
    }
  },

  // 删除待办事项
  delete: async (req, res, next) => {
    try {
      const todo = await Todo.findOneAndDelete({
        _id: req.params.id,
        userId: req.user.id
      });
      
      if (!todo) {
        return res.status(404).json({ error: 'Todo not found' });
      }
      
      res.json({ message: '删除成功' });
    } catch (error) {
      next(error);
    }
  }
};

module.exports = todoController;
```

#### `backend/src/middlewares/auth.js`

```javascript
const jwt = require('jsonwebtoken');

const auth = async (req, res, next) => {
  try {
    // 从请求头获取 token
    const token = req.header('Authorization')?.replace('Bearer ', '');
    
    if (!token) {
      return res.status(401).json({ error: '未授权，请提供 token' });
    }
    
    // 验证 token
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    
    // 将用户信息添加到请求对象
    req.user = decoded;
    
    next();
  } catch (error) {
    res.status(401).json({ error: '未授权，token 无效' });
  }
};

module.exports = auth;
```

on:
  push:
    branches: [ main ]

jobs:
  test:
    runs-on: ubuntu-latest
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Setup Node.js
      uses: actions/setup-node@v3

        npm test

    runs-on: ubuntu-latest

    - uses: actions/checkout@v3

      uses: docker/setup-buildx-action@v2
      
    - name: Login to Docker Hub
      uses: docker/login-action@v2
      with:
        username: ${{ secrets.DOCKER_USERNAME }}
        password: ${{ secrets.DOCKER_PASSWORD }}
        
    - name: Build and push Docker image
      uses: docker/build-push-action@v4
      with:
        context: ./backend
        push: true
        tags: your-dockerhub-username/todo-backend:latest
        
    - name: Deploy to server
      uses: appleboy/ssh-action@v1.0.0
      with:
        host: ${{ secrets.SERVER_HOST }}
        username: ${{ secrets.SERVER_USER }}
        key: ${{ secrets.SERVER_SSH_KEY }}

          docker pull your-dockerhub-username/todo-backend:latest

## 核心技巧

### 1. 让 Claude Code 生成"完整的项目架构"

❌ **不好**：只说"帮我搭建一个 Web 项目"
✅ **好**：明确说出技术栈、功能需求、部署方式

```
请帮我生成完整的 Web 项目架构，需求：
1. 前端：React + TypeScript + Tailwind CSS
2. 后端：Node.js + Express + MongoDB
3. 功能：用户注册/登录、待办事项的 CRUD
4. 部署：用 Docker + GitHub Actions CI/CD
```

---

### 2. 前后端分离，别混在一起

❌ **不好**：前后端代码写在一个项目里
✅ **好**：前后端分离（frontend/ + backend/）

**好处**：
- 独立部署
- 技术栈独立升级
- 团队协作更方便

---

### 3. 用 Docker Compose 管理本地开发环境

❌ **不好**：手动启动 MongoDB、后端、前端
✅ **好**：用 Docker Compose 一键启动

```bash
# 一键启动所有服务
docker-compose up

# 一键停止所有服务
docker-compose down
```

---

### 4. 用 GitHub Actions 自动化 CI/CD

❌ **不好**：手动测试、构建、部署
✅ **好**：用 GitHub Actions 自动化

**工作流**：
1. 代码推送到 main 分支
2. 自动运行测试（Jest）
3. 测试通过后，自动构建 Docker 镜像
4. 自动部署到生产服务器
